-- ===================================================================== -- Hourly budget exhaustion | dt x brand x campaign x hour (IST) -- Hourly spend = campaign debits from the ledger (DEBIT_CPC + PROMO_FALLBACK_DRAW) -- Daily budget = per campaign-day from adtech_platform_campaigns_change_log, -- forward-filled (source: campaign_daily_budget.sql, Statement 1). -- Shows daily_budget, hourly spend, cumulative spend, % used, and -- exhausted_at_hour = first hour cumulative spend >= daily budget. -- Window: trailing 7 days (IST). Add a brand/campaign filter for a focused look. -- ===================================================================== WITH win AS ( SELECT DATE_SUB(CURRENT_DATE('Asia/Kolkata'), INTERVAL 7 DAY) AS start_d, CURRENT_DATE('Asia/Kolkata') AS end_d ), bp AS ( SELECT CURRENT_DATE('Asia/Kolkata') AS report_end_date, (SELECT start_d FROM win) AS min_output_date -- budget spine floor = window start ), -- ---------- daily budget builder (from campaigns change-log) ---------- raw AS ( SELECT id AS campaign_id, advertiser_id, SAFE_CAST(daily_budget AS NUMERIC) AS daily_budget, COALESCE(updated_at, _metadata_timestamp) AS event_ts, GREATEST(DATE(COALESCE(updated_at, _metadata_timestamp), 'Asia/Kolkata'), (SELECT min_output_date FROM bp)) AS event_date, _metadata_log_file, _metadata_log_position, CASE WHEN _metadata_read_method = 'mysql-backfill-fulldump' THEN 0 ELSE 1 END AS source_rank, DATE(TIMESTAMP_SECONDS(SAFE_CAST(start_date AS INT64)), 'Asia/Kolkata') AS campaign_start_date, DATE(TIMESTAMP_SECONDS(SAFE_CAST(end_date AS INT64)), 'Asia/Kolkata') AS campaign_end_date FROM `datapipelineproduction.datos_banco_lake.adtech_platform_campaigns_change_log` WHERE _PARTITIONTIME IS NOT NULL AND _metadata_deleted = FALSE ), ordered AS ( SELECT *, ROW_NUMBER() OVER (PARTITION BY campaign_id, event_date ORDER BY event_ts DESC, source_rank DESC, _metadata_log_file DESC, _metadata_log_position DESC) AS rn_day_desc FROM raw ), daily_close AS ( SELECT campaign_id, advertiser_id, event_date, daily_budget, campaign_start_date, campaign_end_date FROM ordered WHERE rn_day_desc = 1 ), bounds AS ( SELECT campaign_id, GREATEST((SELECT min_output_date FROM bp), COALESCE(LEAST(MIN(campaign_start_date), MIN(event_date)), MIN(event_date))) AS spine_start, LEAST((SELECT report_end_date FROM bp), COALESCE(MAX(campaign_end_date), (SELECT report_end_date FROM bp))) AS spine_end FROM daily_close GROUP BY 1 ), spine AS ( SELECT b.campaign_id, d AS budget_date FROM bounds b, UNNEST(GENERATE_DATE_ARRAY(b.spine_start, GREATEST(b.spine_start, b.spine_end))) AS d ), bud_joined AS ( SELECT s.campaign_id, s.budget_date, dc.daily_budget, dc.advertiser_id FROM spine s LEFT JOIN daily_close dc ON dc.campaign_id = s.campaign_id AND dc.event_date = s.budget_date ), daily_budget AS ( SELECT CAST(campaign_id AS STRING) AS campaign_id, budget_date, SAFE_CAST(NULLIF( COALESCE(LAST_VALUE(daily_budget IGNORE NULLS) OVER fwd, FIRST_VALUE(daily_budget IGNORE NULLS) OVER bwd), 0) AS FLOAT64) AS daily_budget FROM bud_joined WINDOW fwd AS (PARTITION BY campaign_id ORDER BY budget_date ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW), bwd AS (PARTITION BY campaign_id ORDER BY budget_date ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) ), -- ---------- hourly campaign spend from the ledger ---------- led AS ( SELECT DATE(created_at, 'Asia/Kolkata') AS dt, CAST(brand_id AS STRING) AS brand_id, CAST(linked_campaign_id AS STRING) AS campaign_id, EXTRACT(HOUR FROM DATETIME(created_at, 'Asia/Kolkata')) AS hr, SAFE_CAST(amount AS FLOAT64) AS amount FROM `datapipelineproduction.datos_banco_lake.adtech_platform_wallet_ledger_change_log`, win WHERE _metadata_deleted = false AND txn_type IN ('DEBIT_CPC','PROMO_FALLBACK_DRAW') AND DATE(created_at, 'Asia/Kolkata') BETWEEN win.start_d AND win.end_d QUALIFY ROW_NUMBER() OVER (PARTITION BY ledger_txn_id ORDER BY _metadata_timestamp DESC) = 1 ), hourly AS ( SELECT dt, brand_id, campaign_id, hr, ROUND(SUM(amount), 2) AS hour_spend FROM led GROUP BY dt, brand_id, campaign_id, hr ), spend_joined AS ( SELECT h.dt, h.brand_id, h.campaign_id, h.hr, h.hour_spend, db.daily_budget, ROUND(SUM(h.hour_spend) OVER ( PARTITION BY h.dt, h.campaign_id ORDER BY h.hr ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW), 2) AS cumulative_spend FROM hourly h LEFT JOIN daily_budget db ON db.campaign_id = h.campaign_id AND db.budget_date = h.dt ), exhaust AS ( SELECT dt, campaign_id, MIN(IF(daily_budget > 0 AND cumulative_spend >= daily_budget, hr, NULL)) AS exhausted_at_hour FROM spend_joined GROUP BY dt, campaign_id ), brand AS ( SELECT brand_id, brand_name FROM `datapipelineproduction.datos_pocos.product_master` QUALIFY ROW_NUMBER() OVER (PARTITION BY brand_id ORDER BY brand_id) = 1 ) SELECT j.dt, j.brand_id, COALESCE(br.brand_name, CONCAT('advertiser_', j.brand_id)) AS brand_name, j.campaign_id, j.daily_budget, j.hr AS hour, j.hour_spend, j.cumulative_spend, ROUND(100 * SAFE_DIVIDE(j.cumulative_spend, NULLIF(j.daily_budget, 0)), 1) AS pct_budget_used, e.exhausted_at_hour, (j.hr = e.exhausted_at_hour) AS is_exhaustion_hour FROM spend_joined j LEFT JOIN exhaust e ON e.dt = j.dt AND e.campaign_id = j.campaign_id LEFT JOIN brand br ON CAST(br.brand_id AS STRING) = j.brand_id ORDER BY j.dt, j.brand_id, j.campaign_id, j.hr;
